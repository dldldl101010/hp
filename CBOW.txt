import numpy as np
import matplotlib.pyplot as plt
from tensorflow import keras
from tensorflow.keras import layers

words=["I","love","deep","learning"]

words_to_id = {}
for i in range(len(words)):
  words_to_id[words[i]] = i
print("word mapping:", words_to_id)

X = []
y = []

for i in range(1, len(words) - 1):
  context = [words_to_id[words[i - 1]],
             words_to_id[words[i + 1]]]
  target = words_to_id[words[i]]

  X.append(context)
  y.append(target)

  print("input:", X)
  print("output:", y)

model = keras.Sequential([
    layers.Embedding(input_dim=len(words), output_dim=8),
    layers.GlobalAveragePooling1D(),
    layers.Dense(len(words), activation="softmax")
])

model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy"
)

model.fit(np.array(X), np.array(y), epochs=100, verbose=0)

test = np.array([[
    words_to_id["I"],
    words_to_id["deep"]
]])


result = model.predict(test, verbose=0)

predicted_id = np.argmax(result)
print("Predicted Word:", words[predicted_id])
